% Image fixe de l'animation rsjs-ecarts. Parabole des moindres carrés P* (trait plein)
% et parabole écartée P* + s (t-u)(t-v), s = 0,6, u = -1, v = 1,5 (tirets).
% À gauche, abscisses variées : P*(t) = 0,3654 t^2 - 0,0909 t + 0,8773, E(P*) = 1,027,
% E(P* + 0,6 Q) = 9,708. À droite, deux abscisses : P*(t) = 0,56 t + 1,76 (droite des
% moyennes), E = 1,54 pour toute valeur de s.
\begin{tikzpicture}
    \definecolor{oymteal}{RGB}{58,143,143}
    \definecolor{oymorange}{RGB}{184,93,63}
    \pgfplotsset{rsjs/.style={
        width=7cm, height=5.6cm, xmin=-2.5, xmax=2.7, ymin=-0.5, ymax=4.5,
        xtick={-2,-1,0,1,2}, ytick={0,1,2,3,4}, grid=major, grid style={black!12},
        xlabel={$t$}, ylabel={$x$}, samples=120, domain=-2.5:2.7, clip=true,
        title style={font=\small}}}
    \begin{axis}[rsjs, title={Abscisses variées : solution unique}]
        \addplot[very thick, oymteal] {0.36541*x^2 - 0.09094*x + 0.87731};
        \addplot[thick, dashed, oymorange] {0.36541*x^2 - 0.09094*x + 0.87731 + 0.6*(x+1)*(x-1.5)};
        \foreach \t/\x in {-2/2.8, -1.4/1.1, -0.8/1.6, -0.2/0.7, 0.4/1.2, 1/0.8, 1.6/2.0, 2.2/2.3} {
            \edef\tmp{\noexpand\draw[black!55, thick] (axis cs:\t,\x) -- (axis cs:\t,{0.36541*\t*\t - 0.09094*\t + 0.87731});}\tmp
        }
        \addplot[only marks, mark=*, mark size=1.8pt] coordinates {(-2,2.8) (-1.4,1.1) (-0.8,1.6) (-0.2,0.7) (0.4,1.2) (1,0.8) (1.6,2.0) (2.2,2.3)};
        \node[anchor=north west, font=\footnotesize, fill=white, inner sep=2pt, align=left] at (axis cs:-2.4,4.4) {$E(P^*) = 1{,}03$\\ \textcolor{oymorange}{$E = 9{,}71$ pour $s = 0{,}6$}};
    \end{axis}
    \begin{axis}[rsjs, xshift=7.6cm, title={Deux abscisses : infinité de solutions}]
        \draw[black!45, densely dotted] (axis cs:-1,-0.5) -- (axis cs:-1,4.5);
        \draw[black!45, densely dotted] (axis cs:1.5,-0.5) -- (axis cs:1.5,4.5);
        \addplot[very thick, oymteal] {0.56*x + 1.76};
        \addplot[thick, dashed, oymorange] {0.56*x + 1.76 + 0.6*(x+1)*(x-1.5)};
        \addplot[thick, dashed, oymorange] {0.56*x + 1.76 - 0.6*(x+1)*(x-1.5)};
        \addplot[only marks, mark=*, mark size=1.8pt] coordinates {(-1,0.6) (-1,1.4) (-1,1.0) (-1,1.8) (1.5,2.0) (1.5,3.1) (1.5,2.4) (1.5,2.9)};
        \node[font=\footnotesize, anchor=south west] at (axis cs:-1,-0.45) {$u$};
        \node[font=\footnotesize, anchor=south west] at (axis cs:1.5,-0.45) {$v$};
        \node[anchor=north west, font=\footnotesize, fill=white, inner sep=2pt] at (axis cs:-2.4,4.4) {$E = 1{,}54$ pour tout $s$};
    \end{axis}
\end{tikzpicture}
