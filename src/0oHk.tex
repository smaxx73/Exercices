\contenu{
\texte{À l'aide d'un tableur, calculer les quantités suivantes. Dans $\mathcal N(\mu,\sigma)$, le second paramètre est l'écart-type ; dans $\mathcal E(\lambda)$, $\lambda$ est le taux.}
\begin{enumerate}
\item \question{$\mathbb P(X\leq1{,}2)$ pour $X\sim\mathcal N(0,1)$.}
\reponse{\texttt{=LOI.NORMALE.STANDARD.N(1,2;VRAI)} donne $0{,}884930$.
L'argument \texttt{VRAI} demande la fonction de répartition, c'est-à-dire la probabilité à gauche de la valeur indiquée.}
\item \question{$\mathbb P(X\leq102)$ pour $X\sim\mathcal N(120,4)$.}
\reponse{\texttt{=LOI.NORMALE.N(102;120;4;VRAI)} donne $3{,}39767\times10^{-6}$, affiché \texttt{3,39767E-06}. Cette valeur très faible est cohérente : $102$ est à $4{,}5$ écarts-types sous la moyenne.}
\item \question{Le quantile $t$ tel que $\mathbb P(X\leq t)=0{,}9$ pour $X\sim\mathcal N(120,4)$.}
\reponse{\texttt{=LOI.NORMALE.INVERSE.N(0,9;120;4)} donne $t\simeq125{,}1262$.}
\item \question{$\mathbb P(90\leq X\leq120)$ pour $X\sim\mathcal N(100,15)$.}
\reponse{\texttt{=LOI.NORMALE.N(120;100;15;VRAI)-LOI.NORMALE.N(90;100;15;VRAI)} donne $0{,}656296$.}
\item \question{$\mathbb P(X\geq5)$ pour $X\sim\mathcal E(\frac{1}{2})$.}
\reponse{\texttt{=1-LOI.EXPONENTIELLE.N(5;0,5;VRAI)} donne $e^{-2{,}5}\simeq0{,}082085$.}
\item \question{$\mathbb P(X=3)$ et $\mathbb P(X\geq2)$ pour $X\sim\mathcal P(1{,}5)$.}
\reponse{\texttt{=LOI.POISSON.N(3;1,5;FAUX)} donne $0{,}125511$.
L'argument \texttt{FAUX} demande ici la probabilité ponctuelle $\mathbb P(X=3)$.
Pour « au moins deux », utiliser $1-\mathbb P(X\leq1)$ : \texttt{=1-LOI.POISSON.N(1;1,5;VRAI)} donne $0{,}442175$.}
\item \question{Le réel positif $t$ tel que $\mathbb P(100-t\leq X\leq100+t)=0{,}95$ pour $X\sim\mathcal N(100,15)$.}
\reponse{\texttt{=LOI.NORMALE.INVERSE.N(0,975;0;15)} donne $t\simeq29{,}39946$.}
\item \question{\emph{Prolongement -- loi de Student.} Pour $T\sim St(15)$, déterminer $a$ tel que $\mathbb P(T>a)=0{,}05$, puis $c>0$ tel que $\mathbb P(-c\leq T\leq c)=0{,}95$. Pourquoi les probabilités à fournir à la fonction inverse sont-elles différentes ?}
\indication{Pour le premier seuil, convertir la probabilité à droite en probabilité à gauche. Pour l'intervalle central, répartir les $5\%$ restants entre les deux queues, en utilisant la symétrie.}
\reponse{\texttt{=LOI.STUDENT.INVERSE.N(0,95;15)} donne $a=t_{15;0{,}95}\simeq1{,}753050$.
\par Pour l'intervalle central, chaque queue contient $2{,}5\%$ : \texttt{=LOI.STUDENT.INVERSE.N(0,975;15)} donne $c=t_{15;0{,}975}\simeq2{,}131450$. L'intervalle est donc environ $[-2{,}131450;2{,}131450]$. Les probabilités à gauche sont $0{,}95$ et $0{,}975$.}
\item \question{\emph{Prolongement -- loi du khi-deux.} Pour $Q\sim\chi^2(16)$, déterminer $b$ tel que $\mathbb P(Q>b)=0{,}05$. Pour $V\sim\chi^2(15)$, déterminer $\ell$ et $h$ tels que $\mathbb P(V<\ell)=\mathbb P(V>h)=0{,}025$. Quelle probabilité contient $[\ell;h]$ ? Cet intervalle est-il symétrique autour de l'espérance de $V$ ?}
\indication{Utiliser les quantiles à gauche d'ordres $0{,}95$, $0{,}025$ et $0{,}975$. La loi du khi-deux n'est pas symétrique ; son espérance vaut son nombre de degrés de liberté.}
\reponse{\texttt{=LOI.KHIDEUX.INVERSE(0,95;16)} donne $b=q_{16;0{,}95}\simeq26{,}296228$.
\par\texttt{=LOI.KHIDEUX.INVERSE(0,025;15)} donne $\ell\simeq6{,}262138$.
\par\texttt{=LOI.KHIDEUX.INVERSE(0,975;15)} donne $h\simeq27{,}488393$.
\par L'intervalle contient $95\%$ de la probabilité, avec deux queues de même probabilité. Il n'est pas symétrique autour de $\mathbb E(V)=15$ : les distances sont environ $8{,}7379$ et $12{,}4884$. Des queues de même probabilité n'impliquent pas des bornes équidistantes de l'espérance.}
\item \question{\emph{Prolongement -- loi de Fisher.} Pour $F\sim F(15,9)$, déterminer $f$ tel que $\mathbb P(F>f)=0{,}05$. Contrôler le résultat avec la fonction de répartition.}
\indication{Les deux paramètres de la loi de Fisher sont, dans l'ordre, les degrés de liberté du numérateur et du dénominateur.}
\reponse{\texttt{=INVERSE.LOI.F.N(0,95;15;9)} donne $f=F_{15,9;0{,}95}\simeq3{,}006102$.
\par Contrôle : \texttt{=LOI.F.DROITE(f;15;9)}, ou \texttt{=1-LOI.F.N(f;15;9;VRAI)}, renvoie $0{,}05$.
\par Inverser l'ordre des paramètres donne la loi $F(9,15)$, dont le quantile d'ordre $0{,}95$ vaut environ $2{,}5876$.}
\end{enumerate}
}
