\contenu{
\texte{$\mathcal N(\mu,\sigma)$ désigne la loi normale de moyenne $\mu$ et d'écart-type $\sigma$.}
\texte{Une machine de production fabrique des pièces en grande série. Lorsque la machine est bien réglée, la proportion de pièces défectueuses est $p_0=0{,}05$. Lorsqu'elle est déréglée, cette proportion passe à $p_1=0{,}10$.

Pour vérifier l'état de la machine, on prélève un échantillon aléatoire de $n=100$ pièces dans la production, et on observe $k_{\mathrm{obs}}=8$ pièces défectueuses. On modélise les états des pièces par des variables $X_1,\ldots,X_n$ indépendantes de loi $\mathcal B(1,p)$, avec $X_i=1$ si la pièce $i$ est défectueuse, 0 sinon.

\begin{minipage}{\linewidth}
On pose
\[
K_n=\sum_{i=1}^nX_i,\qquad F_n=\frac{K_n}{n},
\qquad f_{\mathrm{obs}}=\frac{k_{\mathrm{obs}}}{n}=0{,}08.
\]
\end{minipage}\par
Pour $n=100$, on abrège $K=K_{100}$ et $F=F_{100}$. On souhaite tester
\[
\begin{cases}
H_0:p=p_0=0{,}05 &\text{(la machine est bien réglée)},\\
H_1:p=p_1=0{,}10 &\text{(la machine est déréglée)}.
\end{cases}
\]
On utilise d'abord l'approximation normale, puis on compare avec le modèle binomial exact. Les calculs normaux sont effectués sans correction de continuité.}
\begin{enumerate}
\question{\item Justifier que, sous $H_0$, la statistique de test
\[
T_n=\frac{F_n-p_0}{\sqrt{\frac{p_0(1-p_0)}{n}}}
\]
suit approximativement une loi $\mathcal N(0,1)$ pour $n$ assez grand.}
\reponse{Les indicatrices $X_i$ ont pour espérance $p$ et pour variance $p(1-p)$. Par linéarité de l'espérance et indépendance,
\[
\mathbb E(F_n)=\frac1n\sum_{i=1}^n\mathbb E(X_i)=p,
\qquad
\operatorname{Var}(F_n)=\frac1{n^2}\sum_{i=1}^n\operatorname{Var}(X_i)
=\frac{p(1-p)}{n}.
\]
Le théorème central limite donne, pour $p\in]0,1[$ fixé,
\[
\frac{F_n-p}{\sqrt{\frac{p(1-p)}{n}}}
\xrightarrow[n\to\infty]{\mathcal L}\mathcal N(0,1).
\]
Sous $H_0$, $p=p_0$, donc $T_n$ a une loi approximativement normale centrée réduite lorsque $n$ est grand. $T_n$ est une variable aléatoire ; la valeur calculée avec $f_{\mathrm{obs}}$ est sa réalisation $t_{\mathrm{obs}}$.

Pour $n=100$, les deux repères usuels sont
\[
np_0=100\times0{,}05=5,\qquad
n(1-p_0)=100\times0{,}95=95.
\]
Le premier est à la limite du repère $np_0\geq5$. Ces repères ne garantissent pas la précision de l'approximation, notamment dans les queues de distribution. La comparaison avec le modèle binomial exact permettra ici de mesurer l'écart sur le risque de première espèce.
}
\question{\item On note $\pi$ la valeur critique : on rejette $H_0$ si $F_n>\pi$. En utilisant la définition du risque de première espèce, exprimer ce risque en fonction de $\pi$, $p_0$ et $n$, exactement puis avec l'approximation normale.}
\indication{Rappel : le risque de première espèce est la probabilité de rejeter $H_0$ sous $H_0$. Standardiser la fréquence avec sa variance sous $p_0$.}
\indication{On peut calculer $\sqrt{\frac{0{,}05\times0{,}95}{100}}\simeq0{,}0218$ avec un tableur. Pour la formule exacte, $K_n$ est entier : utiliser $\lfloor n\pi\rfloor$.}
\reponse{Le risque de première espèce réel est
\begin{align*}
\alpha_{\mathrm{réel}}(\pi)
&=\PP_{p_0}(F_n>\pi)\\
&=\PP_{p_0}\left(\frac{F_n-p_0}{\sqrt{\frac{p_0(1-p_0)}{n}}}
>\frac{\pi-p_0}{\sqrt{\frac{p_0(1-p_0)}{n}}}\right)\\
&\simeq\PP\left(Z>\frac{\pi-p_0}{\sqrt{\frac{p_0(1-p_0)}{n}}}\right),
\qquad Z\sim\mathcal N(0,1)\\
&=1-\Phi\left(\frac{\pi-p_0}{\sqrt{\frac{p_0(1-p_0)}{n}}}\right).
\end{align*}
La dernière expression définit le risque normal approché $\alpha_{\mathrm{approx}}(\pi)$. L'étape qui remplace $T_n$ par $Z$ est une approximation.

Pour $p_0=0{,}05$ et $n=100$,
\[
\alpha_{\mathrm{approx}}(\pi)
=1-\Phi\left(\frac{\pi-0{,}05}{\sqrt{\frac{0{,}05\times0{,}95}{100}}}\right)
\simeq1-\Phi\left(\frac{\pi-0{,}05}{0{,}0218}\right).
\]
Pour le calcul exact, $K_n\sim\mathcal B(n,p_0)$ sous $H_0$, et
\[
\alpha_{\mathrm{réel}}(\pi)=\PP_{p_0}(K_n>n\pi)
=1-\PP_{p_0}(K_n\leq\lfloor n\pi\rfloor).
\]
La fréquence est discrète : on ne peut généralement pas obtenir exactement tout niveau demandé avec une règle de seuil sans randomisation.
}
\question{\item Pour un seuil nominal $\alpha=0{,}05$, déterminer la valeur critique normale $\pi$ et en déduire les régions de non-rejet et de rejet en termes de fréquence observée. Comparer avec une règle binomiale exacte de niveau au plus $5\%$.}
\indication{On pourra utiliser $\Phi(1{,}645)\simeq0{,}95$, où $\Phi$ désigne la fonction de répartition de la loi $\mathcal N(0,1)$.}
\indication{Avec un tableur : \texttt{=0,05+1,645*RACINE(0,05*0,95/100)}. Conserver assez de décimales avant de déterminer le seuil entier de $K$.}
\reponse{On calibre d'abord le risque normal approché à $5\%$ :
\begin{align*}
1-\Phi\left(\frac{\pi-0{,}05}{\sqrt{\frac{0{,}05\times0{,}95}{100}}}\right)=0{,}05
&\quad\Longleftrightarrow\quad
\Phi\left(\frac{\pi-0{,}05}{\sqrt{\frac{0{,}05\times0{,}95}{100}}}\right)=0{,}95\\
&\quad\Longleftrightarrow\quad
\frac{\pi-0{,}05}{\sqrt{\frac{0{,}05\times0{,}95}{100}}}
=\Phi^{-1}(0{,}95).
\end{align*}
Comme $\Phi^{-1}(0{,}95)\simeq1{,}64485$,
\begin{align*}
\pi&=0{,}05+\Phi^{-1}(0{,}95)\sqrt{\frac{0{,}05\times0{,}95}{100}}\\
&\simeq0{,}05+1{,}64485\times0{,}0217945\\
&\simeq0{,}0858488.
\end{align*}
En arrondissant le quantile et l'écart-type, le calcul $0{,}05+1{,}645\times0{,}0218\simeq0{,}0859$, donne une lecture approchée, soit environ $8{,}6\%$.

\begin{minipage}{\linewidth}
La règle normale donne une région de non-rejet $f_{\mathrm{obs}}\leq\pi$ et une région de rejet $f_{\mathrm{obs}}>\pi$. Pour $n=100$, $100\pi\simeq8{,}58488$ ; on rejette donc pour $k_{\mathrm{obs}}\geq9$.

Son risque binomial réel vaut
\[
\PP_{p_0}(K\geq9)\simeq0{,}0630896,
\]
soit $6{,}309\%$, supérieur au seuil nominal de $5\%$.
\end{minipage}

\begin{minipage}{\linewidth}
Le plus petit seuil entier de rejet de niveau au plus $5\%$ est 10, car
\[
\PP_{p_0}(K\geq9)>0{,}05,
\qquad\PP_{p_0}(K\geq10)\simeq0{,}0281883\leq0{,}05.
\]
La règle exacte rejette pour $k_{\mathrm{obs}}\geq10$ et ne rejette pas pour $k_{\mathrm{obs}}\leq9$. Son risque est d'environ $2{,}819\%$.
\end{minipage}
}
\question{\item Au vu des données ($f_{\mathrm{obs}}=0{,}08$), peut-on conclure que la machine est bien réglée au seuil nominal $\alpha=5\%$ ? Calculer également la $p$-valeur normale approchée, puis la $p$-valeur exacte.}
\indication{Avec un tableur, \texttt{=(0,08-0,05)/RACINE(0,05*0,95/100)} donne environ $1{,}37649$. La $p$-valeur normale se calcule par \texttt{=1-LOI.NORMALE.STANDARD.N(1,37649;VRAI)}.}
\reponse{On observe $f_{\mathrm{obs}}=\frac8{100}=0{,}08<\pi\simeq0{,}0858488$. La fréquence est dans la région de non-rejet du test normal. De même, huit défauts sont inférieurs aux deux seuils entiers 9 et 10 : les deux règles ne rejettent pas $H_0$.

\textbf{Conclusion.} Les données ne permettent pas, à ce seuil, de conclure à un dérèglement. Ce non-rejet ne prouve pas que la machine est bien réglée ; il ne donne pas une probabilité de $95\%$ que $H_0$ soit vraie.

\textbf{$p$-valeur normale approchée.} Comme l'alternative augmente la proportion de défauts, on utilise la queue droite. La réalisation de la statistique vaut
\[
t_{\mathrm{obs}}=\frac{0{,}08-0{,}05}{\sqrt{\frac{0{,}05\times0{,}95}{100}}}
\simeq1{,}37649.
\]
La $p$-valeur approchée est
\begin{align*}
p_{\mathrm{val,normal}}
&=1-\Phi(t_{\mathrm{obs}})\\
&\simeq1-\Phi(1{,}37649)\simeq0{,}0843343.
\end{align*}
On obtient l'arrondi $0{,}084$, soit environ $8{,}4\%$, supérieur à $5\%$.

\begin{minipage}{\linewidth}
\textbf{$p$-valeur exacte.} Sous $H_0$, $K\sim\mathcal B(100,0{,}05)$, donc
\[
p_{\mathrm{val,exact}}=\PP_{p_0}(K\geq8)
=1-\PP_{p_0}(K\leq7)\simeq0{,}127960.
\]
Excel : \texttt{=1-LOI.BINOMIALE.N(7;100;0,05;VRAI)}.
Cette valeur est également supérieure à $5\%$ et confirme le non-rejet par le test exact.
\end{minipage}
}
\question{\item Calculer le risque de seconde espèce $\beta$, c'est-à-dire la probabilité de ne pas rejeter $H_0$ alors que la machine est effectivement déréglée. Donner également la puissance. Distinguer l'approximation normale et les risques binomiaux des deux règles entières.}
\indication{Le risque de seconde espèce est $\beta=\PP_{p_1}(\text{ne pas rejeter }H_0)$. Utiliser la variance sous $p_1$, puis la zone de non-rejet de chaque règle.}
\indication{Avec un tableur : \texttt{=RACINE(0,10*0,90/100)} donne $0{,}03$. Avec le seuil arrondi, \texttt{=(0,0859-0,10)/0,03} donne $-0{,}47$, puis \texttt{=LOI.NORMALE.STANDARD.N(-0,47;VRAI)} donne environ $0{,}319$.}
\reponse{Le risque de seconde espèce de la règle normale est
\[
\beta_{\mathrm{réel,normal}}=\PP_{p_1}(F\leq\pi).
\]
Sous $H_1$, $p=p_1=0{,}10$ ; la fréquence a pour moyenne $0{,}10$ et pour écart-type
\[
\sigma_1=\sqrt{\frac{0{,}10\times0{,}90}{100}}
=\sqrt{0{,}0009}=0{,}03.
\]
La variable $\frac{F-0{,}10}{0{,}03}$ est approximativement normale centrée réduite. Ainsi,
\begin{align*}
\PP_{p_1}(F\leq\pi)
&=\PP_{p_1}\left(\frac{F-0{,}10}{0{,}03}
\leq\frac{\pi-0{,}10}{0{,}03}\right)\\
&\simeq\Phi\left(\frac{\pi-0{,}10}{0{,}03}\right)\\
&\simeq\Phi\left(\frac{0{,}0858487537-0{,}10}{0{,}03}\right)\\
&\simeq\Phi(-0{,}471708)\simeq0{,}318568.
\end{align*}
Le calcul normal donne $\beta_{\mathrm{approx}}\simeq0{,}318568$ et une puissance approchée $1-\beta_{\mathrm{approx}}\simeq0{,}681432$.

Avec le seuil arrondi $0{,}0859$, on retrouve le raisonnement
\[
\Phi\left(\frac{0{,}0859-0{,}10}{0{,}03}\right)
=\Phi(-0{,}47)=1-\Phi(0{,}47)\simeq0{,}319.
\]
Si $p=0{,}10$, le calcul normal donne environ $32\%$ de risque de ne pas détecter le dérèglement, soit une puissance d'environ $68\%$.

\begin{minipage}{\linewidth}
\textbf{Risque binomial réel de la règle normale.} Le rejet à partir de neuf défauts correspond au non-rejet pour $K\leq8$. Sous $p_1$, $K\sim\mathcal B(100,0{,}10)$, donc
\[
\beta_{\mathrm{réel,normal}}=\PP_{p_1}(K\leq8)\simeq0{,}320874,
\qquad 1-\beta_{\mathrm{réel,normal}}\simeq0{,}679126.
\]
Excel : \texttt{=LOI.BINOMIALE.N(8;100;0,10;VRAI)}.
\end{minipage}

\begin{minipage}{\linewidth}
\textbf{Risque de la règle exacte de niveau au plus $5\%$.} Le rejet à partir de dix défauts correspond au non-rejet pour $K\leq9$ :
\[
\beta_{\mathrm{exact}}=\PP_{p_1}(K\leq9)\simeq0{,}451290,
\qquad 1-\beta_{\mathrm{exact}}\simeq0{,}548710.
\]
Excel : \texttt{=LOI.BINOMIALE.N(9;100;0,10;VRAI)}.
La règle exacte diminue le risque de première espèce, mais réduit ici la puissance : elle détecte le dérèglement dans environ $54{,}87\%$ des échantillons sous $p_1$.
\end{minipage}
}
\question{\item On souhaite désormais dimensionner le plan de contrôle. Quelle taille d'échantillon $n$ permettrait de rejeter $H_0$ avec une fréquence observée exactement égale à $f_{\mathrm{obs}}=0{,}08$, au seuil nominal de $5\%$ ? Comparer la règle normale et le test exact. Interpréter le résultat : pourquoi augmenter la taille de l'échantillon ?}
\indication{Avec un tableur : \texttt{=0,0475*(1,645/0,03)\^{}2} donne environ $142{,}8$. Arrondir la borne vers le haut, puis vérifier qu'un nombre entier de défauts peut produire exactement $8\%$.}
\reponse{\textbf{Calcul normal théorique.} On cherche $n$ tel qu'une fréquence de $8\%$ soit dans la région critique. Avec $u=\Phi^{-1}(0{,}95)$, la condition de rejet est
\begin{align*}
\frac{0{,}08-0{,}05}{\sqrt{\frac{0{,}05\times0{,}95}{n}}}>u
&\quad\Longleftrightarrow\quad
\frac{0{,}03}{\sqrt{\frac{0{,}0475}{n}}}>u\\
&\quad\Longleftrightarrow\quad
0{,}03>u\sqrt{\frac{0{,}0475}{n}}\\
&\quad\Longleftrightarrow\quad
\left(\frac{0{,}03}{u}\right)^2>\frac{0{,}0475}{n}\\
&\quad\Longleftrightarrow\quad
n>\frac{0{,}0475}{\left(\frac{0{,}03}{u}\right)^2}
=\frac{0{,}0475u^2}{(0{,}03)^2}.
\end{align*}
La borne vaut
\[
\frac{0{,}0475\,\Phi^{-1}(0{,}95)^2}{0{,}0009}\simeq142{,}79257.
\]
Le plus petit entier satisfaisant ce calcul théorique est donc \textbf{143}.

\begin{minipage}{\linewidth}
\textbf{Compatibilité avec une fréquence exactement égale à $8\%$.} Le nombre observé de défauts doit vérifier
\[
k_{\mathrm{obs}}=0{,}08n=\frac{2n}{25}\in\mathbb N.
\]
Comme 2 et 25 sont premiers entre eux, $n$ doit être un multiple de 25. À $n=143$, $0{,}08n=11{,}44$ n'est pas entier. Le premier multiple de 25 supérieur à $142{,}79257$ est donc \textbf{150}, avec $k_{\mathrm{obs}}=12$ défauts. C'est la première taille compatible pour la règle normale.
\end{minipage}

\begin{minipage}{\linewidth}
\textbf{Comparaison avec le test exact.} Pour chaque taille compatible, on calcule la $p$-valeur unilatérale
\[
\PP_{p_0}(K_n\geq0{,}08n),\qquad K_n\sim\mathcal B(n,0{,}05).
\]
Les tailles 25, 50, 75, 100 et 125 donnent toutes une $p$-valeur supérieure à $5\%$. Les suivantes donnent :
\begin{center}
\begin{tabular}{c|ccc}
$n$ & 150 & 175 & 200 \\ \hline
$k_{\mathrm{obs}}=0{,}08n$ & 12 & 14 & 16 \\
$p$-valeur exacte & $0{,}07400$ & $0{,}05711$ & $0{,}04436$
\end{tabular}
\end{center}
La première taille compatible permettant le rejet exact à $5\%$ est donc \textbf{200}, avec 16 défauts. Pour ce dernier calcul, Excel donne
\texttt{=1-LOI.BINOMIALE.N(15;200;0,05;VRAI)}.
\end{minipage}

\textbf{Interprétation.} Avec $n=100$, une fréquence de $8\%$ ne conduit pas au rejet à $5\%$. Lorsque $n$ augmente, l'écart-type de la fréquence, $\sqrt{\frac{p_0(1-p_0)}{n}}$, diminue ; un même écart observé à $p_0$ devient plus significatif. Un plus grand échantillon permet donc de détecter des différences plus faibles, au prix d'un contrôle plus coûteux.

\begin{minipage}{\linewidth}
Le calcul suppose, à chaque taille envisagée, une fréquence exactement égale à $8\%$ ; il ne prédit pas les observations futures. La taille doit être planifiée avant les données : poursuivre le prélèvement jusqu'à obtenir un rejet demanderait une méthode adaptée aux observations successives.
\end{minipage}
}
