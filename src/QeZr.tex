\contenu{
\texte{$\mathcal N(\mu,\sigma)$ désigne la loi normale de moyenne $\mu$ et d'écart-type $\sigma$. Les quantiles $q_{\nu;\gamma}$ de $\chi^2(\nu)$, $t_{\nu;\gamma}$ de $St(\nu)$ et $F_{\nu_1,\nu_2;\gamma}$ de $F(\nu_1,\nu_2)$ sont définis avec une probabilité $\gamma$ à gauche.}
\texte{Dans une expérience fictive sur la dispersion d'impacts, on relève l'\textbf{écart horizontal signé} à une position de référence, en cm : positif vers la droite, négatif vers la gauche. Il ne s'agit pas d'une distance radiale au centre.

On suppose les 25 observations indépendantes de même loi $\mathcal N(\mu,\sigma)$. La limite de dispersion fixée pour cet exercice est $\sigma_0=8$ cm ; elle ne représente pas un référentiel officiel. On souhaite étayer une dispersion inférieure à cette limite, puis examiner le centrage horizontal.

14,2 & 5,7 & 7,2 & 10,1 & 8,9 & 12,5 & 6,3 & &

Pour vérification : $n=25$, $\bar x=9{,}268$ cm, $s^2=9{,}82310$ cm$^2$ et $s\simeq3{,}13418$ cm, avec variance corrigée. Les lettres minuscules désignent les valeurs observées ; $\overline X$, $S^2$ et $S$ désignent les statistiques aléatoires correspondantes.}

Les fonctions \texttt{=MOYENNE(A2:A26)} et \texttt{=VAR.S(A2:A26)} donnent la moyenne et la variance corrigée, sans arrondi intermédiaire.}

\item \question{On veut étayer une dispersion inférieure à 8 cm. Formuler les hypothèses, préciser le caractère unilatéral du test et justifier la statistique et sa loi de référence.}
\reponse{
\textbf{Hypothèses :}
\[
H_0:\sigma\ge8\quad\text{contre}\quad H_1:\sigma<8.
\]
Comme $\sigma>0$, ces hypothèses sont équivalentes à $H_0:\sigma^2\ge64$ contre $H_1:\sigma^2<64$. Le rejet permettra d'étayer une dispersion inférieure à la limite fictive. Le test est \textbf{unilatéral à gauche} : ce sont les petites variances observées qui soutiennent $H_1$.

\textbf{Statistique :} pour $X_1,\ldots,X_n$ indépendantes de loi $\mathcal N(\mu,\sigma)$,
\[
S^2=\frac1{n-1}\sum_{i=1}^n(X_i-\overline X)^2,
\qquad \frac{(n-1)S^2}{\sigma^2}\sim\chi^2(n-1).
\]
On utilise donc
\[
Q=\frac{(n-1)S^2}{\sigma_0^2}=\frac{24S^2}{64}.
\]
\textbf{Loi sous l'hypothèse nulle composite :} à la frontière $\sigma=8$, $Q\sim\chi^2(24)$ exactement sous le modèle normal. Pour $\sigma>8$,
\[
Q=\frac{\sigma^2}{64}\,U,\qquad U\sim\chi^2(24).
\]
Si $c=q_{24;0{,}05}$, alors
\[
\mathbb P_\sigma(Q<c)=\mathbb P\left(U<\frac{64c}{\sigma^2}\right)
\le\mathbb P(U<c)=0{,}05.
\]
Le risque de rejet sous $H_0$ est maximal à la frontière $\sigma=8$. On peut donc calibrer le test à cette frontière ; la loi $\chi^2(24)$ n'est pas la loi de $Q$ pour toutes les valeurs de $H_0$.
}
\indication{Commencer par formuler ce que l'on souhaite \emph{établir} dans $H_1$. Utiliser le pivot $\frac{(n-1)S^2}{\sigma^2}$ et distinguer la frontière $\sigma=\sigma_0$ du reste de l'hypothèse nulle.}

\item \question{Effectuer ce test au seuil de $5\%$ et interpréter le résultat.}
\reponse{
\textbf{Statistique observée :} les données donnent exactement
\[
\bar x=\frac{231{,}7}{25}=9{,}268,\qquad
s^2=\frac{\sum x_i^2-25\bar x^2}{24}
=\frac{2383{,}15-25\times9{,}268^2}{24}=9{,}82310.
\]
Ainsi,
\[
q_{\mathrm{obs}}=\frac{24s^2}{64}
=\frac{235{,}7544}{64}=3{,}6836625.
\]
\textbf{Région critique :} au seuil de $5\%$, on rejette si $q_{\mathrm{obs}}<q_{24;0{,}05}$, avec
\[
q_{24;0{,}05}\simeq13{,}848425.
\]
Comme $3{,}6836625<13{,}848425$, on rejette $H_0$.

\textbf{$p$-valeur :} la probabilité de queue gauche, évaluée à la frontière de $H_0$ où elle est maximale, vaut
\[
p_{\mathrm{val}}=\mathbb P(U\le q_{\mathrm{obs}})
\simeq5{,}86553\times10^{-7},\qquad U\sim\chi^2(24).
\]
\textbf{Conclusion :} sous le modèle normal et l'hypothèse d'indépendance, les données étayent une dispersion horizontale inférieure à 8 cm. Ce rejet ne constitue pas une garantie absolue et ne renseigne pas sur le centrage.

Le test opposé $H_0:\sigma\le8$ contre $H_1:\sigma>8$ chercherait une dispersion excessive, dans la queue droite. Son non-rejet ne suffirait pas à établir $\sigma<8$. Le choix de la direction doit donc répondre à l'objectif annoncé, avant observation.
}
\indication{Pour les calculs suivants, saisir en D1 \texttt{=NB(A2:A26)}, en D2 \texttt{=MOYENNE(A2:A26)}, en D3 \texttt{=VAR.S(A2:A26)}, en D4 \texttt{=RACINE(D3)} et en D5 la valeur $8$. Puis :
\begin{itemize}
\item D6, statistique observée : \texttt{=(D1-1)*D3/D5\^{}2}.
\item Seuil gauche : \texttt{=LOI.KHIDEUX.INVERSE(0,05;D1-1)}.
\item $p$-valeur : \texttt{=LOI.KHIDEUX.N(D6;D1-1;VRAI)}.
\end{itemize}
L'argument \texttt{VRAI} demande la fonction de répartition, donc la probabilité à gauche, et non la densité. Conserver les valeurs non arrondies dans les cellules.}

\item \question{Construire l'intervalle de confiance bilatéral à $95\%$ de $\sigma$ et une borne supérieure unilatérale à $95\%$. Interpréter.}
\reponse{
\textbf{Inversion du pivot :} sous le modèle normal,
\[
\mathbb P\left(q_{24;0{,}025}\le\frac{24S^2}{\sigma^2}\le q_{24;0{,}975}\right)=0{,}95.
\]
Tous les termes sont positifs. En isolant $\sigma^2$, l'intervalle aléatoire devient
\[
I_{\sigma^2}=\left[\frac{24S^2}{q_{24;0{,}975}}\ ;\
\frac{24S^2}{q_{24;0{,}025}}\right].
\]
\textbf{Intervalle observé pour la variance :} les quantiles à gauche sont
\[
q_{24;0{,}025}\simeq12{,}401150,\qquad
q_{24;0{,}975}\simeq39{,}364077.
\]
D'où
\[
I_{\sigma^2,\mathrm{obs}}=
\left[\frac{235{,}7544}{q_{24;0{,}975}}\ ;\
\frac{235{,}7544}{q_{24;0{,}025}}\right]
\simeq[5{,}989075\ ;\ 19{,}010688]\ \text{cm}^2.
\]
Le plus grand quantile figure au dénominateur de la borne inférieure.

\textbf{Intervalle pour l'écart-type :} la racine carrée étant croissante,
\[
I_{\sigma,\mathrm{obs}}\simeq[\sqrt{5{,}989075}\ ;\ \sqrt{19{,}010688}]
\simeq[2{,}44726\ ;\ 4{,}36012]\ \text{cm}.
\]
\textbf{Borne supérieure unilatérale :} puisque $\mathbb P(U\ge q_{24;0{,}05})=0{,}95$, la borne supérieure est
\[
B=\sqrt{\frac{24S^2}{q_{24;0{,}05}}},\qquad
b_{\mathrm{obs}}=\sqrt{\frac{235{,}7544}{13{,}848425\ldots}}
\simeq4{,}12600\ \text{cm}.
\]
La procédure donne l'intervalle unilatéral $]0;B]$, qui couvre $\sigma$ avec une probabilité $0{,}95$ sous le modèle. La borne observée est inférieure à 8 cm, en accord avec le test unilatéral à $5\%$.

\textbf{Interprétation :} le niveau de confiance décrit la proportion d'intervalles couvrant le paramètre fixe lors de répétitions de l'échantillonnage. Les bornes observées expriment l'incertitude sur la dispersion ; elles ne sont pas des limites garanties ni un scénario de « pire cas ». L'intervalle bilatéral et la borne unilatérale ont des constructions différentes ; pour le test unilatéral de la question précédente, c'est la borne unilatérale à $95\%$ qui fournit la correspondance directe.
}
\indication{Avec les cellules précédentes :
\begin{itemize}
\item Borne basse de $\sigma$ :\par
\texttt{=RACINE((D1-1)*D3/LOI.KHIDEUX.INVERSE(0,975;D1-1))}.
\item Borne haute de $\sigma$ :\par
\texttt{=RACINE((D1-1)*D3/LOI.KHIDEUX.INVERSE(0,025;D1-1))}.
\item Borne supérieure unilatérale :\par
\texttt{=RACINE((D1-1)*D3/LOI.KHIDEUX.INVERSE(0,05;D1-1))}.
\end{itemize}
Pour les bornes de la variance, utiliser les mêmes divisions sans la racine carrée.}

\item \question{Une faible dispersion garantit-elle le centrage ? Tester $\mu=0$ contre $\mu\ne0$ à $5\%$ et commenter.}
\reponse{
\textbf{Dispersion et centrage :} $\sigma$ décrit la dispersion autour de la moyenne ; $\mu$ décrit la position moyenne par rapport à la référence. Un petit écart-type peut donc accompagner une moyenne éloignée de zéro.

\textbf{Hypothèses :}
\[
H_0:\mu=0\quad\text{contre}\quad H_1:\mu\ne0.
\]
Le test est bilatéral : le décalage peut être positif ou négatif. La variance est inconnue ; avec les observations normales indépendantes, on utilise
\[
T=\frac{\overline X-0}{\frac{S}{\sqrt n}}\sim St(n-1)=St(24)\quad\text{sous }H_0.
\]
\textbf{Calcul et décision :} l'erreur-type observée vaut $\frac{s}{\sqrt{25}}\simeq0{,}6268365$ cm. Ainsi,
\[
t_{\mathrm{obs}}=\frac{9{,}268}{\frac{\sqrt{9{,}82310}}{5}}\simeq14{,}785355.
\]
Le seuil bilatéral à $5\%$ est $t_{24;0{,}975}\simeq2{,}063899$. Comme $|t_{\mathrm{obs}}|>2{,}063899$, on rejette $H_0$. Avec $V\sim St(24)$,
\[
p_{\mathrm{val}}=\mathbb P(|V|\ge|t_{\mathrm{obs}}|)
=2\mathbb P(V\ge14{,}785355)\simeq1{,}48531\times10^{-13}.
\]
\textbf{Conclusion :} les données étayent un décalage moyen horizontal positif, estimé à $9{,}268$ cm. Une faible dispersion ne garantit donc pas le centrage.

Ces mesures ne déterminent pas la cause du décalage : il faudrait examiner le protocole de mesure et disposer d'informations complémentaires. Elles ne décrivent pas non plus la position dans les deux dimensions. Une distance radiale au centre, toujours positive, ne pourrait pas être interprétée comme cet écart signé pour tester le centrage par $\mu=0$.
}
\indication{Avec les cellules précédentes, calculer en F1 \texttt{=D4/RACINE(D1)} (erreur-type), puis en F2 \texttt{=D2/F1} (statistique observée). Utiliser \texttt{=LOI.STUDENT.INVERSE.BILATERALE(0,05;D1-1)} pour le seuil et \texttt{=LOI.STUDENT.BILATERALE(ABS(F2);D1-1)} pour la $p$-valeur. Cette dernière fonction est déjà bilatérale : ne pas multiplier à nouveau par 2.}

\item \question{Un second échantillon indépendant de 20 écarts signés, normal, a moyenne $8{,}1$ cm et variance corrigée $15{,}3$ cm$^2$. Comparer les variances au seuil de $5\%$.}
\reponse{
\textbf{Hypothèses :}
\[
H_0:\sigma_1^2=\sigma_2^2\quad\text{contre}\quad H_1:\sigma_1^2\ne\sigma_2^2.
\]
Il s'agit d'un test bilatéral. L'ordre du rapport est fixé avant observation, avec le groupe 2 au numérateur :
\[
R=\frac{S_2^2}{S_1^2},\qquad
r_{\mathrm{obs}}=\frac{s_2^2}{s_1^2}=\frac{15{,}3}{9{,}82310}\simeq1{,}557553.
\]
\textbf{Loi sous $H_0$ :} la normalité et l'indépendance des observations donnent
\[
\frac{19S_2^2}{\sigma_2^2}\sim\chi^2(19),\qquad
\frac{24S_1^2}{\sigma_1^2}\sim\chi^2(24).
\]
Ces deux variables sont indépendantes. Sous l'égalité des variances,
\[
R=\frac{\frac{S_2^2}{\sigma_2^2}}{\frac{S_1^2}{\sigma_1^2}}
\sim F(19,24).
\]
\textbf{Région critique :} pour répartir $5\%$ dans les deux queues, les quantiles à gauche sont
\[
F_{19,24;0{,}025}\simeq0{,}407777,\qquad
F_{19,24;0{,}975}\simeq2{,}345154.
\]
On rejette si $r_{\mathrm{obs}}$ est inférieur au premier seuil ou supérieur au second. Ici $0{,}407777<1{,}557553<2{,}345154$ : on ne rejette pas $H_0$.

\textbf{$p$-valeur :} avec $W\sim F(19,24)$,
\[
p_{\mathrm{val}}=2\min\{\mathbb P(W\le r_{\mathrm{obs}}),\mathbb P(W\ge r_{\mathrm{obs}})\}
\simeq0{,}303001.
\]
\textbf{Conclusion :} on ne met pas en évidence de différence de dispersion entre les populations avec ces données et ce test. Le non-rejet ne démontre pas l'égalité des variances, ni l'homogénéité de processus qui auraient produit les observations. La moyenne $8{,}1$ cm du second échantillon n'intervient pas dans ce test ; comparer les moyennes demanderait une autre analyse.
}
\indication{Saisir $n_2=20$ en H1 et $s_2^2=15{,}3$ en H2, puis le rapport en H3 par \texttt{=H2/D3}. Utiliser :
\begin{itemize}
\item Seuil bas : \texttt{=INVERSE.LOI.F.N(0,025;H1-1;D1-1)}.
\item Seuil haut : \texttt{=INVERSE.LOI.F.N(0,975;H1-1;D1-1)}.
\item Probabilité à gauche en H4 :\par
\texttt{=LOI.F.N(H3;H1-1;D1-1;VRAI)}.
\item Probabilité à droite en H5 :\par
\texttt{=LOI.F.DROITE(H3;H1-1;D1-1)}.
\item $p$-valeur bilatérale : \texttt{=2*MIN(H4;H5)}.
\end{itemize}
Conserver l'ordre des degrés de liberté, associé au numérateur et au dénominateur, et les valeurs non arrondies.}
\end{enumerate}
