\contenu{
\texte{On prélève indépendamment des pièces dans une production de grande taille. Chaque pièce est défectueuse avec probabilité $p$.}
\begin{enumerate}
\item \question{Sur $n=200$ pièces, $k_{\mathrm{obs}}=24$ sont défectueuses. Construire l'intervalle de confiance bilatéral approché à $95\%$ pour $p$, centré sur la fréquence observée.}
\indication{Définir $F=K/n$ puis $f_{\mathrm{obs}}=24/200$.}
\reponse{$K\sim\mathcal B(n,p)$ ; $F=K/n$ est un estimateur sans biais de $p$. Avec $f_{\mathrm{obs}}=0{,}12$, l'intervalle de Wald vaut
\[\left[f_{\mathrm{obs}}-u_{0{,}025}\sqrt{\frac{f_{\mathrm{obs}}(1-f_{\mathrm{obs}})}{200}};\
f_{\mathrm{obs}}+u_{0{,}025}\sqrt{\frac{f_{\mathrm{obs}}(1-f_{\mathrm{obs}})}{200}}\right]
\simeq[0{,}07496;0{,}16504].\]
Il s'agit d'un intervalle approché, correspondant à environ $[7{,}50\%;16{,}50\%]$.}
\item \question{Planifier la taille d'un nouvel échantillon pour une largeur d'intervalle d'au plus $0{,}04$, au même niveau de confiance. Utiliser la proportion pilote puis le cas le plus défavorable.}
\reponse{La marge souhaitée vaut $0{,}02$. Avec $p_{\mathrm{pilote}}=0{,}12$ :
\[n\geq\frac{u_{0{,}025}^2\times0{,}12\times0{,}88}{0{,}02^2}\simeq1014{,}1451,\]
soit \textbf{1015 pièces}. Pour le pire cas $p=0{,}5$, il faut \textbf{2401 pièces}. La planification pilote suppose une proportion proche de $0{,}12$ et ne garantit pas la largeur future ; le cas $0{,}5$ majore le terme de variance dans cette formule approchée.}
\end{enumerate}
}
