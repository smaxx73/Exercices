\contenu{
\texte{$\mathcal N(\mu,\sigma)$ désigne la loi normale de moyenne $\mu$ et d’écart-type $\sigma$. Les quantiles sont à gauche : $q_{\nu;\gamma}$ pour la loi $\chi^2(\nu)$, avec une probabilité $\gamma$ à gauche du quantile.}
\texte{Les variables $X_1,\ldots,X_5$ sont indépendantes, de loi $\mathcal N(\mu,\sigma)$, $\sigma>0$. On pose
\begin{align*}
T_1&=\frac15\sum_{i=1}^5X_i, & T_2&=\frac{X_1+X_2}{5}+\frac{X_3+X_4+X_5}{4},\\
T_3&=\frac{2X_1+3X_2}{10}+\frac{X_3+2X_4+X_5}{8},\\
U&=\frac{\sum_{i=1}^5(X_i-\mu)^2}{\sigma^2},&V&=\frac{\sum_{i=1}^5(X_i-T_1)^2}{\sigma^2}.
\end{align*}}
\begin{enumerate}
\item \question{Déterminer la loi de $X_1-X_2$.}
\reponse{Par normalité et indépendance, $X_1-X_2\sim\mathcal N(0,\sigma\sqrt2)$ ; sa variance est $2\sigma^2$.}
\item \question{Étudier le biais de $T_1,T_2,T_3$ et comparer l'efficacité des estimateurs sans biais.}
\reponse{Les sommes des poids sont 1, $23/20$, 1 : $T_1,T_3$ sont sans biais et $b(T_2)=3\mu/20$.
\[\operatorname{Var}(T_1)=\frac{\sigma^2}{5},\qquad
\operatorname{Var}(T_3)=\left(\frac4{100}+\frac9{100}+\frac1{64}+\frac4{64}+\frac1{64}\right)\sigma^2=\frac{179}{800}\sigma^2.\]
Comme $1/5<179/800$, $T_1$ est le plus efficace des deux.}
\item \question{Déterminer les lois de $U$ et de $V$.}
\reponse{$U$ est une somme de cinq carrés de normales centrées réduites indépendantes : $U\sim\chi^2(5)$. Le théorème de Fisher donne $V\sim\chi^2(4)$ et l'indépendance entre $V$ et $T_1$.}
\item \question{Déterminer $x$ tel que $\mathbb P(U>x)=0{,}05$.}
\reponse{$x=q_{5;0{,}95}\simeq11{,}0705$.}
\item \question{À partir de $T_1$ et de $V$, construire une variable $Y$ de loi de Student et préciser ses degrés de liberté.}
\indication{Le dénominateur doit utiliser une variable du khi-deux indépendante du numérateur normal.}
\reponse{$Z=\sqrt5(T_1-\mu)/\sigma\sim\mathcal N(0,1)$ est indépendant de $V$. Ainsi
\[Y=\frac{Z}{\sqrt{V/4}}=\frac{T_1-\mu}{S/\sqrt5}\sim St(4),\qquad S^2=\frac14\sum_{i=1}^5(X_i-T_1)^2.\]
$U$ ne convient pas : $U=V+Z^2$, donc $U$ et $Z$ ne sont pas indépendants.}
\end{enumerate}
}
