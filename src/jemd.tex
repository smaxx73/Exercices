\contenu{

\texte{ 	On rappelle qu'une variable $X$ suit une loi géométrique de paramètre $p \in ]0;1[$ si $X$ est à valeurs dans $\N^*=\{1;2;3;...\}$ et si pour tout $k \in \N^*$,
$$\PP(X=k)=p(1-p)^{k-1}$$

On cherche à estimer ce paramètre $p$ à partir d'un échantillon de variables indépendantes et identiquement distribuées. }

\begin{enumerate}
	\item \texte{ On considère un échantillon $(X_1,X_2,X_3,X_4,X_5)$  ayant pour loi mère une loi géométrique de paramètre $p$ et on suppose que la suite  $(3;4;4;2;3)$ constitue une réalisation de cet échantillon. }
	\begin{enumerate}
		\item \question{ Exprimer la fonction de vraisemblance associée à cet échantillon. }
		\reponse{ D'après le cours, la fonction de vraisemblance associée à cet échantillon est donnée par :
		\begin{align*}
			L(p) &= \PP(X_1=3) \times \PP(X_2=4) \times \PP(X_3=4) \\
&\quad\times \PP(X_4=2) \times \PP(X_5=3) \\
			&= p(1-p)^2 \times p(1-p)^3 \times p(1-p)^3 \times p(1-p) \times p(1-p)^2 \\
			&= p^5(1-p)^{11}
		\end{align*}
		}
		\item \question{ En déduire une estimation de $p$ par la méthode du maximum de vraisemblance.  }
		\reponse{\begin{minipage}{\linewidth}
On cherche à maximiser la fonction $L$ sur $]0;1[$. Pour cela, on calcule la dérivée de $L$ :
\begin{align*}
L'(p)&=5p^4(1-p)^{11}-11p^5(1-p)^{10}\\
&=p^4(1-p)^{10}\bigl(5(1-p)-11p\bigr)\\
&=p^4(1-p)^{10}(5-16p).
\end{align*}
Sur $]0;1[$, les facteurs $p^4$ et $(1-p)^{10}$ sont strictement positifs. Ainsi, $L'(p)=0$ si et seulement si $p=\frac5{16}$ ; la dérivée est positive sur $]0;\frac5{16}[$ et négative sur $]\frac5{16};1[$.

La vraisemblance croît donc jusqu'à $\frac5{16}$, puis décroît. Elle admet un unique maximum en $p=\frac5{16}$.

En conclusion, la valeur la plus vraisemblable pour $p$ est
\[
\widehat p_{\mathrm{obs}}=\frac5{16}=0{,}3125.
\]
Il s'agit d'une estimation du paramètre $p$ par la méthode du maximum de vraisemblance.
\end{minipage}}
	\end{enumerate}

	\item \texte{ Afin de déterminer un estimateur de $p$, on considère maintenant un $n$-échantillon $(X_1,...,X_n)$ ayant pour loi mère une loi géométrique de paramètre $p$ et $n$ entiers non nuls $(x_1,...,x_n)$ constituant une réalisation de cet échantillon.  }
	\begin{enumerate}
		\item \question{ Exprimer la fonction de vraisemblance associée à cet échantillon. }
		\reponse{
		\begin{align*}
			L(x_1,...,x_n,p) &= \PP(X_1=x_1) \times ... \times \PP(X_n=x_n) \\
			&= p(1-p)^{x_1-1} \times ... \times p(1-p)^{x_n-1} \\
			&= p^n(1-p)^{x_1+...+x_n-n}
		\end{align*}
		}
		\item \question{ En utilisant la méthode du maximum de vraisemblance, déterminer un estimateur du paramètre $p$. Discuter le cas où tous les $x_i$ valent 1. }
		\indication{Poser $s_x=\sum_{i=1}^n x_i$, factoriser la dérivée et étudier son signe sur $]0;1[$.}
\reponse{On cherche à maximiser la fonction $L$ sur $]0;1[$. On pose $s_x=\sum_{i=1}^n x_i$, de sorte que $L(p)=p^n(1-p)^{s_x-n}$.

Pour cela, on calcule la dérivée de $L$ par rapport à $p$ :
\begin{align*}
L'(p)&=np^{n-1}(1-p)^{s_x-n}\\
&\quad -(s_x-n)p^n(1-p)^{s_x-n-1}\\
&=p^{n-1}(1-p)^{s_x-n-1}\bigl(n(1-p)-(s_x-n)p\bigr)\\
&=p^{n-1}(1-p)^{s_x-n-1}(n-s_xp).
\end{align*}
\textbf{Si $s_x>n$ :} les facteurs $p^{n-1}$ et $(1-p)^{s_x-n-1}$ sont strictement positifs sur $]0;1[$. Le point $\frac n{s_x}$ appartient à cet intervalle. La dérivée est positive avant $\frac n{s_x}$ et négative après ; la vraisemblance admet donc un unique maximum en
\[
\widehat p_{\mathrm{obs}}=\frac n{s_x}=\frac{n}{\sum_{i=1}^n x_i}.
\]
La statistique correspondante est
\[
\widehat P_n=\frac{n}{\sum_{i=1}^n X_i}.
\]
Sa valeur observée fournit l'estimation par maximum de vraisemblance lorsque $s_x>n$.

\textbf{Si $s_x=n$ :} puisque les $x_i$ sont des entiers strictement positifs, cela signifie qu'ils valent tous 1. Dans ce cas, $L(p)=p^n$ est strictement croissante sur $]0;1[$. Aucun maximum n'est atteint sur ce domaine ; le supremum vaut 1 et est approché lorsque $p\to1$.

Il n'existe donc pas d'estimation par maximum de vraisemblance pour cette réalisation dans le modèle initial $p\in]0;1[$. Si l'on étend le modèle à $p=1$, correspondant à la loi concentrée en 1, la même formule fournit l'estimation $\widehat p_{\mathrm{obs}}=1$ et l'EMV existe alors pour toutes les réalisations.}
	\end{enumerate}
\end{enumerate}
}
