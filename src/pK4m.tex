\contenu{
	\texte{
		On considère un RNN simple à un seul état caché, défini par les équations :
		\[
		h_t = \tanh(W_h h_{t-1} + W_x x_t + b_h), \qquad y_t = W_y h_t + b_y
		\]
		avec les dimensions suivantes : entrée $x_t \in \mathbb{R}$, état caché $h_t \in \mathbb{R}^2$, sortie $y_t \in \mathbb{R}$.
		
		Les paramètres du réseau sont :
		\[
		W_x = \begin{pmatrix} 1 \\ -1 \end{pmatrix}, \quad
		W_h = \begin{pmatrix} 0.5 & 0 \\ 0 & 0.5 \end{pmatrix}, \quad
		b_h = \begin{pmatrix} 0 \\ 0 \end{pmatrix}, \quad
		W_y = \begin{pmatrix} 1 & 1 \end{pmatrix}, \quad
		b_y = 0
		\]
		L'état caché initial est $h_0 = \begin{pmatrix} 0 \\ 0 \end{pmatrix}$.
		On traite la séquence d'entrée $x_1 = 1$, $x_2 = 0$, $x_3 = -1$.
	}
	\begin{enumerate}
		\item
		\question{Calculer $h_1$, $h_2$, $h_3$ en appliquant la formule de récurrence pas à pas. On rappelle que $\tanh$ s'applique composante par composante.}
		\reponse{
			On conserve les expressions exactes jusqu'à l'évaluation numérique ; les valeurs affichées sont arrondies à $10^{-6}$.
			\textbf{Pas $t=1$.} En posant $s_1=\tanh(1)$, on obtient
			\[
			h_1 = \tanh\!\left(W_h h_0 + W_x x_1\right)
			= \begin{pmatrix}\tanh(1)\\\tanh(-1)\end{pmatrix}
			= \begin{pmatrix}s_1\\-s_1\end{pmatrix}
			\approx \begin{pmatrix}0.761594\\-0.761594\end{pmatrix}.
			\]
			\textbf{Pas $t=2$.} Comme $x_2=0$,
			\[
			W_h h_1 + W_x x_2
			= \begin{pmatrix}0.5s_1\\-0.5s_1\end{pmatrix}
			\approx \begin{pmatrix}0.380797\\-0.380797\end{pmatrix}.
			\]
			En posant $s_2=\tanh(0.5s_1)$, il vient
			\[
			h_2 = \begin{pmatrix}s_2\\-s_2\end{pmatrix}
			\approx \begin{pmatrix}0.363399\\-0.363399\end{pmatrix}.
			\]
			\textbf{Pas $t=3$.} Comme $x_3=-1$,
			\[
			W_h h_2 + W_x x_3
			= \begin{pmatrix}0.5s_2-1\\-0.5s_2+1\end{pmatrix}
			\approx \begin{pmatrix}-0.818300\\0.818300\end{pmatrix}.
			\]
			En posant $s_3=\tanh(0.5s_2-1)$, on obtient
			\[
			h_3 = \begin{pmatrix}s_3\\-s_3\end{pmatrix}
			\approx \begin{pmatrix}-0.674144\\0.674144\end{pmatrix}.
			\]
		}
		
		\item
		\question{En déduire les sorties $y_1$, $y_2$, $y_3$.}
		\reponse{
			On applique $y_t = W_y h_t + b_y$. Puisque $W_y = \begin{pmatrix}1 & 1\end{pmatrix}$ et $b_y = 0$ :
			\[
			y_t = \begin{pmatrix}1 & 1\end{pmatrix}\begin{pmatrix}s_t\\-s_t\end{pmatrix}
			= s_t-s_t=0.
			\]
			Ainsi, $y_1=y_2=y_3=0$ exactement. L'état initial nul, les paramètres $W_x$ et $W_h=0.5\,I_2$ et l'imparité de $\tanh$ garantissent que les deux composantes de $h_t$ restent opposées.
		}
		
		\item
		\question{Décrire qualitativement ce que fait ce réseau : comment l'état caché $h_t$ évolue-t-il lorsque l'entrée change de signe ?}
		\reponse{
			Les deux composantes de $h_t$ sont en permanence opposées ($h_t^{(2)} = -h_t^{(1)}$), pour trois raisons conjointes :
			\begin{itemize}
				\item $W_x$ injecte $x_t$ dans la première composante et $-x_t$ dans la seconde ;
				\item $W_h = 0.5\,I_2$ atténue l'état précédent de façon identique sur les deux composantes, préservant leur relation d'opposition ;
				\item $\tanh$ est une fonction impaire, donc $\tanh(-u) = -\tanh(u)$ conserve l'opposition.
			\end{itemize}
			En posant $s_t=h_t^{(1)}$, on a donc
			\[
			h_t=\begin{pmatrix}s_t\\-s_t\end{pmatrix},
			\qquad s_t=\tanh(0.5s_{t-1}+x_t), \qquad s_0=0.
			\]
			Le signe de $s_t$ est celui de $0.5s_{t-1}+x_t$ : il dépend de l'entrée courante et d'une mémoire atténuée de l'état précédent. Un changement de signe de l'entrée ne garantit donc pas celui de l'état caché. Par exemple, après $x_1=1$, une entrée $x_2=-0.1$ donnerait
			\[
			s_2=\tanh\!\left(0.5\tanh(1)-0.1\right)>0.
			\]
			Pour la séquence de l'énoncé, l'état reste non nul à $t=2$ malgré $x_2=0$, puis change de signe à $t=3$ parce que l'entrée $x_3=-1$ domine le terme de mémoire. La sortie $y_t$ étant toujours nulle, l'information sur la séquence reste dans l'état caché $h_t$.
		}
		
		\item
		\question{(\textit{Bonus}) Que se passerait-il si on remplaçait $\tanh$ par la fonction identité ? Réécrire alors $h_t$ comme une combinaison linéaire des entrées passées $x_1, \ldots, x_t$.}
		\indication{Développer la récurrence $h_t = W_h h_{t-1} + W_x x_t$ et itérer.}
		\reponse{
			Avec l'activation identité, la récurrence devient $h_t = W_h h_{t-1} + W_x x_t$. Par développement itératif (avec $h_0 = 0$) :
			\[
			h_t = W_h^{t-1}W_x x_1 + W_h^{t-2}W_x x_2 + \cdots + W_h^0 W_x x_t = \sum_{k=1}^{t} W_h^{t-k} W_x x_k.
			\]
			Puisque $W_h = 0.5\,I_2$, on a $W_h^{t-k} = (0.5)^{t-k} I_2$. En substituant :
			\[
			h_t = \sum_{k=1}^{t} (0.5)^{t-k} I_2\, W_x x_k = W_x \left(\sum_{k=1}^{t} (0.5)^{t-k} x_k\right).
			\]
			Le vecteur $h_t$ est proportionnel à $W_x = \begin{pmatrix}1\\-1\end{pmatrix}$, le facteur de proportionnalité étant une somme des entrées passées pondérée par des coefficients $(0.5)^{t-k}$ qui décroissent géométriquement avec leur ancienneté. Il ne s'agit pas d'une moyenne, car ces coefficients ne sont pas normalisés :
			\[
			\sum_{k=1}^{t}(0.5)^{t-k}=2(1-2^{-t}).
			\]
		}
	\end{enumerate}
}
