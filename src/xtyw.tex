\contenu{
\texte{Une entreprise de transport maritime mesure la profondeur de corrosion, en millimètres, sur des coques de navires. On modélise cette profondeur par une variable aléatoire $X$ de loi uniforme sur $[0;\theta]$, où $\theta>0$ est inconnu. Sa densité est
\[
f_\theta(x)=\begin{cases}\frac1\theta&\text{si }0\leq x\leq\theta,\\[2pt] 0&\text{sinon.}\end{cases}
\]
On dispose d'un échantillon $(X_1,\ldots,X_n)$ de mesures indépendantes de même loi que $X$, de valeurs observées $x_1,\ldots,x_n\geq0$. On note $m=\max(x_1,\ldots,x_n)$, supposé strictement positif.}
\begin{enumerate}
\item \question{Exprimer la vraisemblance $L(\theta)=L(x_1,\ldots,x_n,\theta)$ en fonction de $\theta$, en distinguant les cas $\theta<m$ et $\theta\geq m$.}
\indication{La vraisemblance est le produit des densités. Elle est nulle dès qu'une observation sort de $[0;\theta]$.}
\reponse{Par indépendance, $L(\theta)=\prod_{i=1}^nf_\theta(x_i)$. Ce produit est non nul si et seulement si toutes les observations appartiennent à $[0;\theta]$, c'est-à-dire si $\theta\geq m$. Ainsi,
\[
L(\theta)=\begin{cases}0&\text{si }0<\theta<m,\\[2pt] \frac1{\theta^n}&\text{si }\theta\geq m.\end{cases}
\]}
\item \question{Étudier les variations de $L$ sur $]0;+\infty[$ et en déduire l'estimation $\widehat\theta_{\mathrm{obs}}$ du maximum de vraisemblance. Pourquoi ne peut-on pas l'obtenir en annulant la dérivée ?}
\indication{Tracer l'allure de $L$ : elle est nulle, puis saute en $\theta=m$, puis décroît.}
\reponse{$L$ est nulle sur $]0;m[$. Sur $[m;+\infty[$, $L(\theta)=\theta^{-n}$ est strictement décroissante, car sa dérivée $-n\theta^{-n-1}$ est négative. Le maximum est donc atteint au bord du domaine où $L$ est non nulle :
\[
\widehat\theta_{\mathrm{obs}}=m=\max(x_1,\ldots,x_n).
\]
La dérivée ne s'annule nulle part : le maximum est atteint en un point où $L$ n'est pas continue. Ici, le support de la loi dépend du paramètre ; la méthode « annuler la dérivée » ne s'applique pas et il faut étudier directement les variations.}
\item \question{En déduire l'estimateur du maximum de vraisemblance $\widehat\Theta_n$.}
\reponse{La statistique correspondante est
\[
\widehat\Theta_n=\max(X_1,\ldots,X_n),
\]
parfois notée $X_{(n)}$. Elle prend toujours une valeur inférieure ou égale à $\theta$.}
\item \question{On admet que $\mathbb E(\widehat\Theta_n)=\frac n{n+1}\theta$. L'estimateur $\widehat\Theta_n$ est-il sans biais ? Proposer un estimateur sans biais $\widetilde\Theta_n$ de $\theta$.}
\indication{Multiplier $\widehat\Theta_n$ par une constante bien choisie.}
\reponse{Le biais vaut
\[
b(\widehat\Theta_n)=\frac n{n+1}\theta-\theta=-\frac\theta{n+1}<0.
\]
L'estimateur est biaisé : il sous-estime $\theta$ en moyenne, ce qui est cohérent avec $\widehat\Theta_n\leq\theta$. Le biais tend vers $0$ lorsque $n\to\infty$. L'estimateur
\[
\widetilde\Theta_n=\frac{n+1}n\max(X_1,\ldots,X_n)
\]
vérifie $\mathbb E(\widetilde\Theta_n)=\frac{n+1}n\times\frac n{n+1}\theta=\theta$ : il est sans biais.}
\item \question{On observe 5 mesures : $2{,}3$ ; $3{,}1$ ; $2{,}8$ ; $3{,}5$ ; $2{,}6$ mm. Calculer les estimations $\widehat\theta_{\mathrm{obs}}$ et $\widetilde\theta_{\mathrm{obs}}$.}
\reponse{Le maximum des observations vaut $3{,}5$ mm, donc
\[
\widehat\theta_{\mathrm{obs}}=3{,}5\ \mathrm{mm},
\qquad
\widetilde\theta_{\mathrm{obs}}=\frac65\times3{,}5=4{,}2\ \mathrm{mm}.
\]
La première valeur est l'estimation du maximum de vraisemblance ; la seconde est la réalisation d'un estimateur sans biais. Être sans biais est une propriété de l'estimateur, en moyenne sur les échantillons : cela ne garantit pas que $4{,}2$ mm soit plus proche de $\theta$ que $3{,}5$ mm pour cet échantillon.}
\end{enumerate}
}
